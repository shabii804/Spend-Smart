   536→  DateTime _parseAnchorMonth(String key) {
   537→    // key format: "year-month-day" (e.g., "2026-10-7")
   538→    final parts = key.split('-');
   539→    if (parts.length >= 2) {
   540→      return DateTime(int.parse(parts[0]), int.parse(parts[1]), 1);
   541→    }
   542→    return DateTime.now();
   543→  }
   544→
   545→  /// Round up to a "nice" maximum for the y-axis: 1k, 2k, 5k, 10k, etc.
   546→  double _niceMax(double v) {
   547→    if (v <= 0) return 1000;
   548→    if (v <= 500) return 500;
   549→    if (v <= 1000) return 1000;
   550→    if (v <= 2000) return 2000;
   551→    if (v <= 3000) return 3000;
   552→    if (v <= 5000) return 5000;
   553→    if (v <= 10000) return 10000;
   554→    final pow10 = math.pow(10, (math.log(v) / math.log(10)).ceil());
   555→    return (v / pow10).ceil() * pow10.toDouble();
   556→  }

   568→class _BreakdownDonut extends StatefulWidget {

   573→  @override

   576→
   577→class _BreakdownDonutState extends State<_BreakdownDonut> {
   578→  bool _showData = false;
   579→
   580→  @override
   581→  void initState() {
   582→    super.initState();
   583→    Future.delayed(const Duration(milliseconds: 240), () {
   584→      if (mounted) setState(() => _showData = true);
   585→    });
   586→  }
   587→
   588→  @override
   589→  Widget build(BuildContext context) {
   590→    if (widget.items.isEmpty) {
   591→      return GlassCard(
   592→        padding: const EdgeInsets.all(20),
   593→        child: Text(AppStrings.noDataYet,
   594→            style: TextStyles.body(context,
   595→                color: AppColorsOf.secondaryText(context))),
   596→      );
   597→    }
   598→    final safeTotal = widget.total == 0 ? 1.0 : widget.total;
   599→    final sections = <PieChartSectionData>[];
   600→    for (int i = 0; i < widget.items.length; i++) {
   601→      final entry = widget.items[i];
   602→      final c = entry.key;
   603→      final pct = entry.value / safeTotal;
   604→      // Only show the % label on slices that are big enough (>= 6%).
   605→      final showLabel = pct >= 0.06;
   606→      sections.add(PieChartSectionData(
   607→        color: c.color,
   608→        // 0 on first render, real value after _showData flips → arcs sweep in
   609→        value: _showData ? entry.value : 0.0001,
   610→        title: showLabel ? '${(pct * 100).round()}%' : '',
   611→        titleStyle: const TextStyle(
   612→          color: Colors.white,
   613→          fontSize: 11,
   614→          fontWeight: FontWeight.w700,
   615→        ),
   616→        titlePositionPercentageOffset: 0.62,
