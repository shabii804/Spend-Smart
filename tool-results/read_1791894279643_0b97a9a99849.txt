   424→  // Top categories — horizontal scroll of small cards

   426→  Widget _topCategoriesScroll(BuildContext context, List<MapEntry<Category, double>> items) {
   427→    if (items.isEmpty) {
   428→      return GlassCard(
   429→        padding: const EdgeInsets.all(16),
   430→        child: Text(
   431→          AppStrings.noDataYet,
   432→          style: TextStyles.body(context,
   433→              color: AppColorsOf.secondaryText(context)),
   434→        ),
   435→      );
   436→    }
   437→    // Use total spent as the denominator so the percentages reflect each
   438→    // category's share of overall spending (not of the top category).
   439→    final totalSpent = items.fold<double>(0.0, (s, e) => s + e.value);
   440→    final safeTotal = totalSpent == 0 ? 1.0 : totalSpent;
   441→    return SizedBox(
   442→      height: 120,
   443→      child: ListView.separated(
   444→        scrollDirection: Axis.horizontal,
   445→        itemCount: items.length,
   446→        separatorBuilder: (_, __) => const SizedBox(width: 12),
   447→        itemBuilder: (_, i) {
   448→          final c = items[i].key;
   449→          final amt = items[i].value;
   450→          final pct = ((amt / safeTotal) * 100).round();
   451→          return Container(
   452→            width: 168,
   453→            padding: const EdgeInsets.all(14),
   454→            decoration: BoxDecoration(
   455→              color: AppColorsOf.card(context),
   456→              border: Border.all(color: AppColorsOf.border(context)),
   457→              borderRadius: BorderRadius.circular(20),
   458→            ),
   459→            child: Column(
   460→              crossAxisAlignment: CrossAxisAlignment.start,
   461→              children: [
   462→                Row(
   463→                  children: [
   464→                    Container(
   465→                      width: 36,
   466→                      height: 36,
   467→                      decoration: BoxDecoration(
   468→                        color: c.color.withOpacity(0.14),
   469→                        shape: BoxShape.circle,
   470→                      ),
   471→                      alignment: Alignment.center,
   472→                      child: Icon(c.icon, color: c.color, size: 18),
   473→                    ),
   474→                    const Spacer(),
   475→                    Container(
   476→                      padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 3),
   477→                      decoration: BoxDecoration(
   478→                        color: c.color.withOpacity(0.14),
   479→                        borderRadius: BorderRadius.circular(8),

   481→                      child: Text('$pct%',
   482→                          style: TextStyles.caption(context,

   494→                        .copyWith(fontFeatures: const [FontFeature.tabularFigures()])),
   495→              ],

   620→/// Parallax wrapper — translates child slower than scroll for depth effect.
   621→/// Uses NotificationListener to track scroll without managing a controller.
   622→class _ParallaxHero extends StatefulWidget {
   623→  final Widget child;
   624→  const _ParallaxHero({required this.child});
   625→
   626→  @override
   627→  State<_ParallaxHero> createState() => _ParallaxHeroState();
   628→}
   629→
   630→class _ParallaxHeroState extends State<_ParallaxHero> {
   631→  double _offset = 0.0;
   632→
   633→  @override
   634→  Widget build(BuildContext context) {
   635→    return NotificationListener<ScrollNotification>(
   636→      onNotification: (notification) {
   637→        if (notification is ScrollUpdateNotification) {
   638→          final pixels = notification.metrics.pixels;
   639→          if (pixels >= 0) {
   640→            setState(() {
   641→              // Parallax depth: hero moves up at 25% of scroll speed
   642→              _offset = (pixels * 0.25).clamp(0.0, 120.0);
   643→            });
   644→          }
   645→        }
   646→        return false;
   647→      },
   648→      child: Transform.translate(
   649→        offset: Offset(0, -_offset),
   650→        child: widget.child,
   651→      ),
   652→    );
   653→  }
   654→}

   656→/// Notifications bell button that wiggles periodically when there are
   657→/// unread notifications. Red dot badge appears with a scale pop.
   658→/// Respects reduce-motion (renders static, no wiggle).
   659→class _BellButton extends StatefulWidget {
   660→  final int unread;
   661→  final VoidCallback onTap;
   662→
   663→  const _BellButton({required this.unread, required this.onTap});
   664→
   665→  @override
   666→  State<_BellButton> createState() => _BellButtonState();
   667→}
   668→
   669→class _BellButtonState extends State<_BellButton>
   670→    with SingleTickerProviderStateMixin {
   671→  late final AnimationController _wiggle;
   672→
   673→  @override
   674→  void initState() {
   675→    super.initState();
   676→    _wiggle = AnimationController(
   677→      vsync: this,
   678→      duration: const Duration(milliseconds: 600),
   679→    );
   680→    // Schedule the first wiggle after 800ms, then repeat every 4 seconds.
   681→    // Only wiggles when there are unread notifications (checked in _scheduleNext).
   682→    _scheduleNext(const Duration(milliseconds: 800));
   683→  }
   684→
   685→  void _scheduleNext(Duration delay) {
   686→    Future.delayed(delay, () {
   687→      if (!mounted) return;
   688→      if (widget.unread > 0 && !AppMotion.reduceMotion(context)) {
   689→        _wiggle.forward(from: 0);
   690→      }
   691→      _scheduleNext(const Duration(seconds: 4));
   692→    });
   693→  }
   694→
   695→  @override
   696→  void dispose() {
   697→    _wiggle.dispose();
   698→    super.dispose();
   699→  }
   700→
   701→  @override
   702→  Widget build(BuildContext context) {
   703→    return Semantics(
   704→      label: 'Notifications${widget.unread > 0 ? ', ${widget.unread} unread' : ''}',
   705→      button: true,
   706→      child: GestureDetector(
   707→        onTap: () {
   708→          HapticFeedback.selectionClick();
   709→          widget.onTap();
   710→        },
   711→        child: Container(
   712→          width: 44,
   713→          height: 44,
   714→          decoration: BoxDecoration(
   715→            color: AppColorsOf.card(context),
   716→            shape: BoxShape.circle,
   717→            border: Border.all(color: AppColorsOf.border(context)),
   718→          ),
   719→          child: Stack(
   720→            alignment: Alignment.center,
   721→            children: [
   722→              // ── Bell icon (rotates back and forth when wiggle fires) ──
   723→              AnimatedBuilder(
   724→                animation: _wiggle,
   725→                builder: (context, child) {
   726→                  // 3 oscillations, ±15°, damped sin wave
   727→                  final t = _wiggle.value;
   728→                  final angle = (t * 6 * 3.14159) * (1 - t) * 0.26;
   729→                  return Transform.rotate(angle: angle, child: child);
   730→                },
   731→                child: Icon(Icons.notifications_none_rounded,
   732→                    color: AppColorsOf.mainText(context), size: 20),
   733→              ),
   734→              // ── Unread badge (scales in when unread > 0) ──
   735→              if (widget.unread > 0)
   736→                Positioned(
   737→                  top: 8,
   738→                  right: 9,
   739→                  child: TweenAnimationBuilder<double>(
   740→                    tween: Tween(begin: 0.0, end: 1.0),
   741→                    duration: AppMotion.base,
   742→                    curve: AppMotion.celebrate,
   743→                    builder: (context, v, child) {
   744→                      return Transform.scale(scale: v, child: child);
   745→                    },
   746→                    child: Container(
   747→                      width: 8,
   748→                      height: 8,
   749→                      decoration: const BoxDecoration(
   750→                        color: AppColors.danger,
   751→                        shape: BoxShape.circle,
   752→                      ),
   753→                    ),
   754→                  ),
   755→                ),
   756→            ],
   757→          ),
   758→        ),
   759→      ),
   760→    );
   761→  }
