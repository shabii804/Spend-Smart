   455→  // Top categories — horizontal scroll of small cards

   457→  Widget _topCategoriesScroll(BuildContext context, List<MapEntry<Category, double>> items) {
   458→    if (items.isEmpty) {
   459→      return GlassCard(
   460→        padding: const EdgeInsets.all(16),
   461→        child: Text(
   462→          AppStrings.noDataYet,
   463→          style: TextStyles.body(context,
   464→              color: AppColorsOf.secondaryText(context)),
   465→        ),
   466→      );
   467→    }
   468→    // Use total spent as the denominator so the percentages reflect each
   469→    // category's share of overall spending (not of the top category).
   470→    final totalSpent = items.fold<double>(0.0, (s, e) => s + e.value);
   471→    final safeTotal = totalSpent == 0 ? 1.0 : totalSpent;
   472→    return SizedBox(
   473→      height: 120,
   474→      child: ListView.separated(
   475→        scrollDirection: Axis.horizontal,
   476→        itemCount: items.length,
   477→        separatorBuilder: (_, __) => const SizedBox(width: 12),
   478→        itemBuilder: (_, i) {
   479→          final c = items[i].key;
   480→          final amt = items[i].value;
   481→          final pct = ((amt / safeTotal) * 100).round();
   482→          return Container(
   483→            width: 168,
   484→            padding: const EdgeInsets.all(14),
   485→            decoration: BoxDecoration(
   486→              color: AppColorsOf.card(context),
   487→              border: Border.all(color: AppColorsOf.border(context)),
   488→              borderRadius: BorderRadius.circular(20),
   489→            ),
   490→            child: Column(
   491→              crossAxisAlignment: CrossAxisAlignment.start,
   492→              children: [
   493→                Row(
   494→                  children: [
   495→                    Container(
   496→                      width: 36,
   497→                      height: 36,
   498→                      decoration: BoxDecoration(
   499→                        color: c.color.withOpacity(0.14),
   500→                        shape: BoxShape.circle,
   501→                      ),
   502→                      alignment: Alignment.center,
   503→                      child: Icon(c.icon, color: c.color, size: 18),
   504→                    ),
   505→                    const Spacer(),
   506→                    Container(
   507→                      padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 3),
   508→                      decoration: BoxDecoration(
   509→                        color: c.color.withOpacity(0.14),
   510→                        borderRadius: BorderRadius.circular(8),
   511→                      ),
   512→                      child: Text('$pct%',
   513→                          style: TextStyles.caption(context,
   514→                              weight: FontWeight.w700).copyWith(color: c.color, fontSize: 11)),

   525→                        .copyWith(fontFeatures: const [FontFeature.tabularFigures()])),
   526→              ],
   527→            ),

   529→        },

   651→/// Parallax wrapper — translates child slower than scroll for depth effect.
   652→/// Uses NotificationListener to track scroll without managing a controller.
   653→class _ParallaxHero extends StatefulWidget {
   654→  final Widget child;
   655→  const _ParallaxHero({required this.child});
   656→
   657→  @override
   658→  State<_ParallaxHero> createState() => _ParallaxHeroState();
   659→}
   660→
   661→class _ParallaxHeroState extends State<_ParallaxHero> {
   662→  double _offset = 0.0;
   663→
   664→  @override
   665→  Widget build(BuildContext context) {
   666→    return NotificationListener<ScrollNotification>(
   667→      onNotification: (notification) {
   668→        if (notification is ScrollUpdateNotification) {
   669→          final pixels = notification.metrics.pixels;
   670→          if (pixels >= 0) {
   671→            setState(() {
   672→              // Parallax depth: hero moves up at 25% of scroll speed
   673→              _offset = (pixels * 0.25).clamp(0.0, 120.0);
   674→            });
   675→          }
   676→        }
   677→        return false;
   678→      },
   679→      child: Transform.translate(
   680→        offset: Offset(0, -_offset),
   681→        child: widget.child,
   682→      ),
   683→    );
   684→  }
   685→}
